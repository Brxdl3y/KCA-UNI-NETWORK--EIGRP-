 GLOBAL ROUTER CONFIGURATION (Applied on all routers)
------------------------------------------------------------
! Enable EIGRP with AS number 100
router eigrp 100
 !
 ! Advertise all relevant networks (example per router below)
 ! Disable auto-summary for precision
 no auto-summary
 !
 ! Enable MD5 authentication (example for interfaces)
 key chain EIGRP-KEY
   key 1
     key-string kcauEIGRPsecure
 !
!
! Passive interface configuration (LAN-facing interfaces)
router eigrp 100
 passive-interface FastEthernet0/1
 passive-interface FastEthernet0/0
!
------------------------------------------------------------
 ROUTER R1 (Technology Dept)
------------------------------------------------------------
hostname TECH-EDGE-R1
enable secret <EIGRP>
!
interface Loopback0
 ip address 1.1.1.1 255.255.255.255
! 
interface GigabitEthernet0/0
 ip address 10.0.2.1 255.255.255.0
 duplex full
 speed 100
 no shutdown
!
interface GigabitEthernet0/1
 ip address 10.0.0.1 255.255.255.0
 duplex auto
 speed auto
 no shutdown
!
interface GigabitEthernet0/2
 ip address 10.0.1.1 255.255.255.0
 duplex full
 speed 100
 no shutdown
!
router eigrp 100
no auto-summary
passive-interface GigabitEthernet0/2
 network 10.0.2.0 0.0.0.255
 network 10.0.3.0 0.0.0.255
 network 10.0.0.0 0.0.0.255
!
